function doPost(e){
  var out, lock=null;
  try{
    var req=JSON.parse((e&&e.postData&&e.postData.contents)||'{}');
    var fn=req.fn, arg=req.arg;
    if(!/^api_/.test(fn) || typeof globalThis[fn]!=='function') throw 'Unknown action';
    if(vers_()['SCHEMA_V']!==SCHEMA_VERSION){ var l0=LockService.getScriptLock(); l0.waitLock(25000); try{ db_(); }finally{ l0.releaseLock(); } }
    if(!READ_ONLY[fn]){ lock=LockService.getScriptLock(); lock.waitLock(25000); }
    out={ r: globalThis[fn](arg) };
  }catch(err){
    out={ e: (err&&err.message)?err.message:String(err) };
  }finally{ if(lock){ try{ lock.releaseLock(); }catch(x){} } }
  return json_(out);
}

function ensureSheets_(ss){
  Object.keys(SCHEMA).forEach(function(name){
    var sh=ss.getSheetByName(name);
    if(!sh){ sh=ss.insertSheet(name); sh.getRange(1,1,1,SCHEMA[name].length).setValues([SCHEMA[name]]); }
    else{
      var have=sh.getRange(1,1,1,Math.max(1,sh.getLastColumn())).getValues()[0].map(String);
      SCHEMA[name].forEach(function(c){ if(have.indexOf(c)<0){ sh.getRange(1,sh.getLastColumn()+1).setValue(c); have.push(c); } });
    }
    sh.setFrozenRows(1);
    try{ sh.getRange(1,1,sh.getMaxRows(),sh.getMaxColumns()).setNumberFormat('@'); }catch(e){}
    bump_(name);
  });
  var def=ss.getSheetByName('Sheet1'); if(def && def.getLastRow()===0) ss.deleteSheet(def);
  if(readAll_('Programs').length===0) DEFAULT_PROGRAMS.forEach(function(p,i){ p.sort=i+1; p.logoId=''; appendRow_('Programs',p); });
  if(!readAll_('Users').some(function(u){return String(u.username).toLowerCase()===ADMIN_USERNAME;})){
    appendRow_('Users',{userId:'u-admin',fullName:ADMIN_NAME,username:ADMIN_USERNAME,passHash:'',salt:Utilities.getUuid(),

  readAll_('Roles').forEach(function(r){ if(r.status==='active') updateWhere_('Roles','roleId',r.roleId,{status:'approved'}); });
  // legacy: expenses before v2 were already paid

function readAll_(name){
  if(_cache[name]) return _cache[name];
  var ver=vers_()['V_'+name]||'0', key='S5:'+name+':'+ver, c=cacheSvc_();
  try{
    var n=c.get(key+':n');
    if(n){ var keys=[]; for(var i=0;i<Number(n);i++) keys.push(key+':'+i); var got=c.getAll(keys), parts=[];
      for(var j=0;j<keys.length;j++){ if(got[keys[j]]==null){ parts=null; break; } parts.push(got[keys[j]]); }
      if(parts) return (_cache[name]=JSON.parse(parts.join('')));
    }
  }catch(e){}
  var sh=sheet_(name), last=sh.getLastRow(), rows=[];
  if(last>=2){
    var idx=headerIdx_(sh), vals=sh.getRange(2,1,last-1,sh.getLastColumn()).getValues();

function updateWhere_(name,keyCol,keyVal,patch){
  var sh=sheet_(name), idx=headerIdx_(sh), last=sh.getLastRow(); if(last<2) return false;
  var ki=idx[keyCol], keys=sh.getRange(2,ki+1,last-1,1).getValues();
  for(var r=0;r<keys.length;r++){
    if(String(keys[r][0])===String(keyVal)){
      var rowR=sh.getRange(r+2,1,1,sh.getLastColumn()), vals=rowR.getValues()[0];
      Object.keys(patch).forEach(function(c){ if(idx[c]!=null && patch[c]!==undefined) vals[idx[c]]=String(patch[c]==null?'':patch[c]); });
      rowR.setValues([vals]); bump_(name); return true;
    }
  }
  return false;
}

}

var REQUIRED_VOL=['fullName','mobile','email','state','city','dob','address','bloodGroup','tshirt','skills','emName','emRelation','emMobile','aadhaarNumber'];
function validateVolunteer_(p,isNew,hasPhoto,hasAadhaarImg){
  var miss=REQUIRED_VOL.filter(function(k){return !String(p[k]||'').trim();});
  if(miss.length) throw 'Please fill all fields (missing: '+miss.join(', ')+').';
  if(digits_(p.mobile).length<10 || digits_(p.emMobile).length<10) throw 'Please enter valid 10-digit mobile numbers.';
  if(String(p.aadhaarNumber).replace(/\D/g,'').length!==12) throw 'Aadhaar number must be 12 digits.';
  if(isNew && !hasPhoto) throw 'Please add your photo.';
  if(isNew && !hasAadhaarImg) throw 'Please upload a photo of your Aadhaar card.';
  if(isNew && !p.consent) throw 'Please accept the Aadhaar consent.';
}
function api_signup(p){
  p=p||{}; var u=String(p.username||'').trim().toLowerCase(), type=p.accountType==='artist'?'artist':'volunteer';
  if(!p.fullName||!u||!p.password) throw 'Please fill name, username and password.';
  if(!/^[a-z0-9._]{3,30}$/.test(u)) throw 'Username: 3-30 small letters, numbers, dot or underscore.';
  if(String(p.password).length<6) throw 'Password must be at least 6 characters.';
  if(digits_(p.mobile).length<10) throw 'Please enter a valid 10-digit mobile number.';
  if(!/^[^@\s]+@[^@\s]+\.[^@\s]+$/.test(String(p.email||'').trim())) throw 'Please enter a valid email.';
  if(!String(p.state||'').trim()||!String(p.city||'').trim()) throw 'Please enter your state and city.';
  if(!p.terms) throw 'Please accept the terms to continue.';
  if(type==='volunteer') validateVolunteer_(p,true,!!p.photo,!!p.aadhaarImage);
  var users=readAll_('Users');
  if(users.some(function(x){return String(x.username).toLowerCase()===u;})) throw 'That username is already taken.';
  if(users.some(function(x){return x.mobile && digits_(x.mobile)===digits_(p.mobile);})) throw 'That mobile number is already registered. Try logging in.';
  if(users.some(function(x){return String(x.email).toLowerCase()===String(p.email).trim().toLowerCase();})) throw 'That email is already registered.';
  var salt=Utilities.getUuid(), id=uid_('u'), row={userId:id,fullName:String(p.fullName).trim(),username:u,passHash:hash_(p.password,salt),salt:salt,
    email:String(p.email).trim(),mobile:digits_(p.mobile),state:p.state||'',city:p.city||'',accountType:type,createdAt:now_(),
    photoId:p.photo?saveFile_(SUB_PHOTOS,p.photo,'profile-'+id+'.jpg'):''};
  if(type==='volunteer'){
    ['dob','address','bloodGroup','tshirt','skills','emName','emRelation'].forEach(function(k){ row[k]=String(p[k]).trim(); });
    row.emMobile=digits_(p.emMobile); row.aadhaarNumber=String(p.aadhaarNumber).replace(/\D/g,'');
    row.aadhaarFileId=saveFile_(SUB_AADHAAR,p.aadhaarImage,'aadhaar-'+id+'.jpg'); row.consentAt=now_();
  }
  appendRow_('Users',row); log_('',{userId:id,fullName:row.fullName},'Account created',type,id);
  return api_login({username:u,password:p.password});
}

function api_login(p){
  p=p||{}; var user=findLoginUser_(p.username);
  if(!user) throw 'No account found with that username, email or mobile.';
  if(T_(user.isDisabled)) throw 'This account is disabled. Contact the organisers.';
  if(!user.passHash) throw 'Password not set yet. Use "Forgot password".';
  if(hash_(p.password,user.salt)!==user.passHash) throw 'Wrong password.';
  if(isAnyAdmin_(user) && !deviceOk_(user,p.device)){
    var code=''+Math.floor(100000+Math.random()*900000), ch=Utilities.getUuid();
    cacheSvc_().put('otp-'+ch,JSON.stringify({u:user.userId,c:code,n:0}),600);
    mail_(user.email,APP_NAME+' — login code '+code,'Hello '+user.fullName+',\n\nYour '+APP_NAME+' login code is: '+code+'\nIt is valid for 10 minutes.\n\nIf this was not you, change your password.');
    return {otp:true,challenge:ch,email:String(user.email).replace(/^(.{2}).*(@.*)$/,'$1***$2')};
  }

function api_verifyOtp(p){
  var c=cacheSvc_(), raw=c.get('otp-'+p.challenge); if(!raw) throw 'Code expired. Please log in again.';
  var o=JSON.parse(raw);
  if(o.n>=5){ c.remove('otp-'+p.challenge); throw 'Too many attempts. Please log in again.'; }
  if(String(p.code).trim()!==o.c){ o.n++; c.put('otp-'+p.challenge,JSON.stringify(o),600); throw 'Wrong code.'; }
  c.remove('otp-'+p.challenge);
  var user=user_(o.u), out=newSession_(user);
  if(p.remember){
    var dev=Utilities.getUuid()+Utilities.getUuid(), list=jp_(user.devicesJSON).filter(function(d){return new Date(d.exp)>new Date();}).slice(-9);
    list.push({h:hash_(dev,pepper_()),exp:new Date(Date.now()+DEVICE_DAYS*864e5).toISOString()});
    updateWhere_('Users','userId',user.userId,{devicesJSON:JSON.stringify(list)}); out.device=dev;
  }
  log_('',user,'Logged in (2-step)','',user.userId);
  return out;
}

function api_changePassword(p){
  var me=userByToken_(p.token);
  if(me.passHash && hash_(p.oldPassword||'',me.salt)!==me.passHash) throw 'Current password is wrong.';
  if(String(p.newPassword||'').length<6) throw 'New password must be at least 6 characters.';
  updateWhere_('Users','userId',me.userId,{passHash:hash_(p.newPassword,me.salt)});
  return true;
}
function api_forgotPassword(p){
  var email=String((p&&p.email)||'').trim().toLowerCase();
  var user=readAll_('Users').filter(function(x){return x.email && String(x.email).toLowerCase()===email;})[0];
  if(!user) throw 'No account with that email.';
  var code=''+Math.floor(100000+Math.random()*900000);
  cacheSvc_().put('reset-'+user.userId,code,900);
  mail_(user.email,APP_NAME+' — password reset code','Hello '+user.fullName+',\n\nYour '+APP_NAME+' reset code is: '+code+'\nIt is valid for 15 minutes.\n\nYour username: '+user.username);
  return {userId:user.userId};
}
function api_resetPassword(p){
  var c=cacheSvc_(), code=c.get('reset-'+p.userId);
  if(!code||code!==String(p.code).trim()) throw 'Wrong or expired code.';
  if(String(p.password||'').length<6) throw 'Password must be at least 6 characters.';
  var user=user_(p.userId); if(!user) throw 'Account not found.';
  updateWhere_('Users','userId',user.userId,{passHash:hash_(p.password,user.salt)});
  c.remove('reset-'+p.userId);
  return newSession_(user);  // email already verified by the code
}

function expiry_(p,ev){
  var m=p.mode||'', n=Date.now(), tz=Session.getScriptTimeZone();
  if(m==='15m') return new Date(n+15*6e4).toISOString();
  if(m==='1h') return new Date(n+60*6e4).toISOString();
  if(m==='today'){ var s=Utilities.formatDate(new Date(),tz,"yyyy-MM-dd")+' 23:59:59'; return Utilities.parseDate(s,tz,'yyyy-MM-dd HH:mm:ss').toISOString(); }
  if(m==='event'){ var e=find_('Events','eventId',ev); var end=(e&&(e.endDate||e.startDate))||''; if(!end) throw 'Event has no end date.'; return Utilities.parseDate(end+' 23:59:59',tz,'yyyy-MM-dd HH:mm:ss').toISOString(); }
  if(m==='off') return '';
  if(m==='custom'){ var mins=num_(p.minutes); if(!(mins>0)) throw 'Enter minutes.'; return new Date(n+mins*6e4).toISOString(); }
  throw 'Choose how long to give access.';
}

function api_requestAccess(p){
  var me=userByToken_(p.token); need_(me,p.eventId,['volunteer']);

  if(readAll_('AccessRequests').some(function(r){return r.eventId===p.eventId&&r.userId===me.userId&&r.section===p.section&&r.status==='pending';})) throw 'Request already sent.';
  appendRow_('AccessRequests',{reqId:uid_('q'),eventId:p.eventId,userId:me.userId,section:p.section,reason:p.reason||'',status:'pending',createdAt:now_()});

var STATUS_TEXT={todo:'To Do',progress:'In Progress',partial:'Partially Done',blocked:'Needs Help',cant:"Can't Complete",done:'Done'};
function api_updateTask(p){
  var me=userByToken_(p.token), t=find_('Tasks','taskId',p.taskId); if(!t) throw 'Task not found.';
  need_(me,t.eventId,['admin','volunteer']);
  if(!canUpdateTask_(me,t)) throw 'This task is assigned to someone else.';

  if(st){
    if(TASK_STATUSES.indexOf(st)<0) throw 'Unknown status.';
    if((st==='cant'||st==='blocked') && !note) throw 'Please write a reason.';
    var pg=st==='done'?100:(st==='todo'?0:num_(p.progress!==undefined?p.progress:t.progress));
    if(st==='partial' && (pg<=0||pg>=100)) throw 'Please choose how much is done (1–99%).';
    patch.status=st; patch.progress=String(pg); patch.statusNote=note; patch.doneAt=st==='done'?now_():'';
  }
  if(p.checklist){ patch.checklistJSON=JSON.stringify(p.checklist.map(function(x){return {text:String(x.text||''),done:!!x.done};})); }
  var files=saveFiles_(p.files,'task-'+t.taskId);

    taskPeople_(t).filter(function(id){return id!==me.userId;}).forEach(function(id){ notify_(t.eventId,id,msg,t.taskId); });
    if(st) log_(t.eventId,me,'Task: '+STATUS_TEXT[st],t.title+(note?' — '+note:''),t.assignedTo||'');
  }
  return api_eventData({token:p.token,eventId:t.eventId});
}

function expOut_(x){
  var o={}; SCHEMA.Expenses.forEach(function(k){o[k]=x[k];});
  o.amount=num_(x.amount); o.settledAmount=num_(x.settledAmount); o.needsAdmin=T_(x.needsAdmin);
  var rq=user_(x.requestedBy||x.addedBy); o.requestedName=rq?rq.fullName:'';
  var ap=x.approvedBy?user_(x.approvedBy):null; o.approvedName=ap?ap.fullName:'';
  var t=x.teamId?find_('Teams','teamId',x.teamId):null; o.teamName=t?t.name:'';
  var v=x.vendorId?find_('Vendors','vendorId',x.vendorId):null; o.vendorName=v?v.name:'';
  o.date=x.spentOn||String(x.addedAt||'').slice(0,10);
  return o;
}
function financeOut_(e){
  var ev=e.eventId, heads=jp_(e.budgetJSON), total=num_(e.budgetTotal), limit=num_(e.approvalLimit)||25000;
  var exps=readAll_('Expenses').filter(function(x){return x.eventId===ev;}).map(expOut_).sort(function(a,b){return String(b.date+b.addedAt).localeCompare(String(a.date+a.addedAt));});
  var inc=readAll_('Income').filter(function(x){return x.eventId===ev;}).map(function(x){ var o={}; SCHEMA.Income.forEach(function(k){o[k]=x[k];}); o.amount=num_(x.amount);
    var a=x.artistId?user_(x.artistId):null; o.artistName=a?a.fullName:''; return o; }).sort(function(a,b){return String(b.receivedOn).localeCompare(String(a.receivedOn));});
  var vendors=readAll_('Vendors').filter(function(v){return v.eventId===ev;}).map(function(v){
    var paid=0, approved=0; exps.forEach(function(x){ if(x.vendorId===v.vendorId){ if(x.status==='paid') paid+=x.amount; else if(x.status==='approved') approved+=x.amount; } });
    return {vendorId:v.vendorId,name:v.name,service:v.service,phone:v.phone,note:v.note,contractAmount:num_(v.contractAmount),paid:paid,approved:approved,due:num_(v.contractAmount)-paid};
  });
  var s={income:0,incomeByKind:{},paid:0,approvedUnpaid:0,requested:0,cash:0,bank:0,byHead:{},byTeam:{},byDay:{},advancesOut:0};
  inc.forEach(function(i){ s.income+=i.amount; s.incomeByKind[i.kind]=(s.incomeByKind[i.kind]||0)+i.amount; s[i.account==='cash'?'cash':'bank']+=i.amount; });
  exps.forEach(function(x){
    if(x.status==='paid'){ s.paid+=x.amount; s[x.account==='cash'?'cash':'bank']-=x.amount;
      s.byHead[x.head]=(s.byHead[x.head]||0)+x.amount; var tn=x.teamName||'No team'; s.byTeam[tn]=(s.byTeam[tn]||0)+x.amount; s.byDay[x.date]=(s.byDay[x.date]||0)+x.amount;
      if(x.kind==='advance') s.advancesOut+=Math.max(0,x.amount-x.settledAmount); }
    else if(x.status==='approved') s.approvedUnpaid+=x.amount;
    else if(x.status==='requested') s.requested+=x.amount;
  });
  var names=heads.map(function(h){return h.name;});
  Object.keys(s.byHead).forEach(function(n){ if(names.indexOf(n)<0){ heads.push({name:n,alloc:0}); names.push(n);} });
  return {total:total,approvalLimit:limit,defaultHeads:BUDGET_HEADS,incomeKinds:INCOME_KINDS,payModes:PAY_MODES,summary:s,
    heads:heads.map(function(h){return {name:h.name,alloc:num_(h.alloc),spent:s.byHead[h.name]||0};}),
    expenses:exps,income:inc,vendors:vendors,
    artists:rolesOf_(ev).filter(function(r){return r.role==='artist'&&r.status==='approved';}).map(function(r){var u=user_(r.userId)||{};return {userId:r.userId,name:u.fullName,adoptedBy:adoptersOf_(ev,r.userId)};})};
}

function budgetWarn_(ev,head,amount){
  var e=find_('Events','eventId',ev), f=financeOut_(e), h=f.heads.filter(function(x){return x.name===head;})[0], w=[];
  if(h && h.alloc>0 && h.spent+amount>h.alloc) w.push(head+' budget will cross by ₹'+Math.round(h.spent+amount-h.alloc));
  if(f.total>0 && f.summary.paid+amount>f.total) w.push('Total budget will cross by ₹'+Math.round(f.summary.paid+amount-f.total));
  return w;
}
function api_requestExpense(p){
  var me=userByToken_(p.token), x=p.expense||{}; need_(me,p.eventId,['admin','volunteer']);
  var amt=num_(x.amount); if(!(amt>0)) throw 'Please enter the amount.';
  if(!x.head) throw 'Please choose a budget head.';
  var kind=['expense','reimbursement','advance'].indexOf(x.kind)>=0?x.kind:'expense';
  var e=find_('Events','eventId',p.eventId), limit=num_(e.approvalLimit)||25000;
  var row={kind:kind,head:x.head,amount:amt,note:x.note||'',payee:x.payee||'',spentOn:x.spentOn||'',teamId:x.teamId||'',vendorId:x.vendorId||'',needsAdmin:amt>limit?'TRUE':'FALSE'};
  if(x.bill) row.billId=saveFile_(SUB_BILLS,x.bill,'bill-'+Date.now()+'.jpg');
  if(x.expId){
    var old=find_('Expenses','expId',x.expId); if(!old||old.eventId!==p.eventId) throw 'Not found.';
    if(!(hasSection_(me,p.eventId,'finance') || (old.requestedBy===me.userId && old.status==='requested'))) throw 'You cannot edit this now.';
    if(old.status==='paid' && roleOf_(me,p.eventId)!=='admin') throw 'Only admins can edit a paid entry.';
    updateWhere_('Expenses','expId',x.expId,row); log_(p.eventId,me,'Expense edited',x.head+' ₹'+amt);
  } else {
    row.expId=uid_('x'); row.eventId=p.eventId; row.status='requested'; row.requestedBy=me.userId; row.addedBy=me.userId; row.addedAt=now_(); row.settledAmount=0;
    appendRow_('Expenses',row); log_(p.eventId,me,'Expense requested',kind+' · '+x.head+' ₹'+amt);
  }
  var d=api_eventData({token:p.token,eventId:p.eventId}); d.warnings=budgetWarn_(p.eventId,x.head,amt); return d;

function api_decideExpense(p){
  var me=userByToken_(p.token), x=find_('Expenses','expId',p.expId); if(!x) throw 'Not found.';
  needSection_(me,x.eventId,'finance');
  if(x.status!=='requested' && !(roleOf_(me,x.eventId)==='admin' && x.status!=='paid')) throw 'This request is already decided.';
  var e=find_('Events','eventId',x.eventId), limit=num_(e.approvalLimit)||25000;
  if(num_(x.amount)>limit && roleOf_(me,x.eventId)!=='admin') throw 'Above ₹'+limit+' — only an admin can decide this.';
  if(!p.approve && !String(p.reason||'').trim()) throw 'Please give a reason for rejecting.';
  updateWhere_('Expenses','expId',x.expId,{status:p.approve?'approved':'rejected',approvedBy:me.userId,approvedAt:now_(),rejectReason:p.approve?'':p.reason});
  log_(x.eventId,me,p.approve?'Expense approved':'Expense rejected',x.head+' ₹'+x.amount+(p.approve?'':' — '+p.reason),x.requestedBy);
  var rq=user_(x.requestedBy); if(rq&&rq.email&&rq.userId!==me.userId) mail_(rq.email,APP_NAME+' — expense '+(p.approve?'approved':'rejected'),'Your request "'+x.head+' ₹'+x.amount+'" for '+e.name+' was '+(p.approve?'approved.':'rejected. Reason: '+p.reason));
  var d=api_eventData({token:p.token,eventId:x.eventId}); d.warnings=p.approve?budgetWarn_(x.eventId,x.head,0):[]; return d;
}

    teams.forEach(function(t){ b.appendListItem(t.name+' — Head: '+(t.head?t.head.fullName+' (@'+t.head.username+')':'-')+(t.members.length?' — Members: '+t.members.map(function(m){return m.fullName;}).join(', '):'')); }); }
  var f=financeOut_(e);
  b.appendParagraph('Finance').setHeading(DocumentApp.ParagraphHeading.HEADING2);
  b.appendParagraph('Income: Rs '+f.summary.income+'   Paid expenses: Rs '+f.summary.paid+'   Net: Rs '+(f.summary.income-f.summary.paid)+'   Cash: Rs '+f.summary.cash+'   Bank: Rs '+f.summary.bank);
  if(f.heads.length){ var br=[['Head','Budget','Spent']]; f.heads.forEach(function(h){ br.push([h.name,String(h.alloc),String(h.spent)]); }); b.appendTable(br); }
  var sch=scheduleOf_(e.eventId);
  if(sch.length){ b.appendParagraph('Schedule').setHeading(DocumentApp.ParagraphHeading.HEADING2); sch.forEach(function(s){ b.appendListItem((s.day||'')+' '+(s.time||'')+' — '+s.title+(s.detail?' ('+s.detail+')':'')); }); }
  doc.saveAndClose();
  return {url:doc.getUrl()};
}

/* ============================================================
 *  DAILY BACKUP (copy of the database sheet, last 14 kept)
 * ========================================================== */
function ensureBackupTrigger_(){
  try{
    var props=PropertiesService.getScriptProperties(); if(props.getProperty('BACKUP_TRIGGER')) return;
    if(!ScriptApp.getProjectTriggers().some(function(t){return t.getHandlerFunction()==='dailyBackup';}))
      ScriptApp.newTrigger('dailyBackup').timeBased().everyDays(1).atHour(2).create();
    props.setProperty('BACKUP_TRIGGER','1');
  }catch(e){}
}
function dailyBackup(){
  var ss=db_(), folder=subFolder_(SUB_BACKUPS);
  DriveApp.getFileById(ss.getId()).makeCopy('Backup '+Utilities.formatDate(new Date(),Session.getScriptTimeZone(),'yyyy-MM-dd'),folder);
  var files=[], it=folder.getFiles(); while(it.hasNext()) files.push(it.next());
  files.sort(function(a,b){return b.getDateCreated()-a.getDateCreated();});
  files.slice(BACKUP_KEEP).forEach(function(f){ f.setTrashed(true); });
  var old=new Date(Date.now()-864e5).toISOString();
  readAll_('Sessions').filter(function(s){return s.exp<old;}).forEach(function(s){ deleteWhere_('Sessions','id',s.id); });
}
