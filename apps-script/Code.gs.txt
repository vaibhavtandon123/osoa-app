  appendRow_('Sessions',{id:uid_('s'),userId:user.userId,tokenHash:hash_(token,pepper_()),exp:new Date(Date.now()+TOKEN_DAYS*864e5).toISOString(),createdAt:now_()});
  updateWhere_('Users','userId',user.userId,{lastLogin:now_()});
  return {token:token,me:meInfo_(user_(user.userId))};
}
function deviceOk_(user,device){
  if(!device) return false; var h=hash_(device,pepper_());
  return jp_(user.devicesJSON).some(function(d){ return d.h===h && new Date(d.exp)>new Date(); });
}
function api_login(p){
  p=p||{}; limit_('login',30,900,'Too many login attempts from this connection. Please wait 15 minutes.');
  var user=findLoginUser_(p.username);
  if(!user) throw 'No account found with that username, email or mobile.';
  if(T_(user.isDisabled)) throw 'This account is disabled. Contact the organisers.';
  if(!user.passHash) throw 'Password not set yet. Use "Forgot password".';
  var fk='fail-'+user.userId, c=cacheSvc_(), fails=Number(c.get(fk)||0);
  if(fails>=5) throw 'Too many wrong passwords. This account is locked for 15 minutes. You can also use "Forgot password".';
  if(!pwOk_(user,p.password)){ c.put(fk,String(fails+1),900); throw fails+1>=5?'Too many wrong passwords. This account is locked for 15 minutes.':'Wrong password. '+(5-fails-1)+' attempt'+(5-fails-1===1?'':'s')+' left.'; }
  c.remove(fk); user=user_(user.userId);
  return otpOrSession_(user,p.device);
}
function otpOrSession_(user,device){
  if(isAnyAdmin_(user) && !deviceOk_(user,device)){
    var code=''+Math.floor(100000+Math.random()*900000), ch=Utilities.getUuid();
    cacheSvc_().put('otp-'+ch,JSON.stringify({u:user.userId,c:code,n:0}),600);
    mail_(user.email,APP_NAME+' — login code '+code,'Hello '+user.fullName+',\n\nYour '+APP_NAME+' login code is: '+code+'\nIt is valid for 10 minutes.\n\nIf this was not you, change your password.');
    return {otp:true,challenge:ch,email:String(user.email).replace(/^(.{2}).*(@.*)$/,'$1***$2')};
  }
  return newSession_(user);
}
function api_verifyOtp(p){
  var c=cacheSvc_(), raw=c.get('otp-'+p.challenge); if(!raw) throw 'Code expired. Please log in again.';
  var o=JSON.parse(raw);
  if(o.n>=5){ c.remove('otp-'+p.challenge); throw 'Too many attempts. Please log in again.'; }
  if(String(p.code).trim()!==o.c){ o.n++; c.put('otp-'+p.challenge,JSON.stringify(o),600); throw 'Wrong code.'; }
  c.remove('otp-'+p.challenge);
  var user=user_(o.u), out=newSession_(user);
  if(p.remember){
    var dev=Utilities.getUuid()+Utilities.getUuid(), list=jp_(user.devicesJSON).filter(function(d){return new Date(d.exp)>new Date();}).slice(-9);
    list.push({h:hash_(dev,pepper_()),exp:new Date(Date.now()+DEVICE_DAYS*864e5).toISOString()});
    updateWhere_('Users','userId',user.userId,{devicesJSON:JSON.stringify(list)}); out.device=dev;
  }
  log_('',user,'Logged in (2-step)','',user.userId);
  return out;
}

function api_resetPassword(p){
  var c=cacheSvc_(), code=c.get('reset-'+p.userId), nk='resetn-'+p.userId, n=Number(c.get(nk)||0);
  if(!code) throw 'This code has expired. Please ask for a new code.';
  if(code!==String(p.code).trim()){ n++; if(n>=5){ c.remove('reset-'+p.userId); c.remove(nk); throw 'Too many wrong codes. Please ask for a new code.'; } c.put(nk,String(n),900); throw 'Wrong code. '+(5-n)+' attempt'+(5-n===1?'':'s')+' left.'; }
  c.remove(nk);
  if(String(p.password||'').length<6) throw 'Password must be at least 6 characters.';
  var user=user_(p.userId); if(!user) throw 'Account not found.';
  updateWhere_('Users','userId',user.userId,{passHash:pwHash_(p.password,user.salt)});
  c.remove('reset-'+p.userId); c.remove('fail-'+user.userId);
  return otpOrSession_(user_(user.userId),p.device);  // admins still need the login code on a new device
}

    var f=DriveApp.getFileById(id), app=appFolder_().getId(), ok=false, ps=f.getParents();

/* ---- Section access (request / grant with timer / revoke) ---- */
function expiry_(p,ev){
  var m=p.mode||'', n=Date.now(), tz=Session.getScriptTimeZone();
  if(m==='15m') return new Date(n+15*6e4).toISOString();
  if(m==='1h') return new Date(n+60*6e4).toISOString();
  if(m==='today'){ var s=Utilities.formatDate(new Date(),tz,"yyyy-MM-dd")+' 23:59:59'; return Utilities.parseDate(s,tz,'yyyy-MM-dd HH:mm:ss').toISOString(); }
  if(m==='event'){ var e=find_('Events','eventId',ev); var end=(e&&(e.endDate||e.startDate))||''; if(!end) throw 'Event has no end date.'; return Utilities.parseDate(end+' 23:59:59',tz,'yyyy-MM-dd HH:mm:ss').toISOString(); }
  if(m==='off') return '';
  if(m==='custom'){ var mins=num_(p.minutes); if(!(mins>0)) throw 'Enter minutes.'; return new Date(n+mins*6e4).toISOString(); }
  throw 'Choose how long to give access.';
}

var STAGE_NAME={registered:'Registered',level1:'Level 1 – Shortlisted',interview:'Interview',level2:'Level 2 – Final',level3:'Level 3 – Confirmed',waitlist:'Waitlist',rejected:'Not selected',withdrawn:'Dropped out'};
function stageOf_(r){ if(r.stage) return r.stage; return r.status==='approved'?'level2':(r.status==='rejected'?'rejected':'registered'); }

function calDate_(s){ return new Date(String(s).length<=10?s+'T00:00:00+05:30':String(s).slice(0,16)+':00+05:30'); }
function calFmt_(d){ return Utilities.formatDate(d,'Asia/Kolkata',"yyyy-MM-dd'T'HH:mm"); }
// occurrences of an entry that start between from and to (Date objects)
function calOccurrences_(e,from,to){
  var st=calDate_(e.start), out=[], until=e.repeatUntil?calDate_(String(e.repeatUntil).slice(0,10)+'T23:59'):null, d=new Date(st.getTime()), n=0;
  if(!e.repeat){ if(st>=from&&st<=to) out.push(st); return out; }
  while(d<=to && n<1000 && (!until||d<=until)){
    if(d>=from) out.push(new Date(d.getTime()));
    if(e.repeat==='daily') d.setDate(d.getDate()+1); else if(e.repeat==='weekly') d.setDate(d.getDate()+7); else if(e.repeat==='monthly') d.setMonth(d.getMonth()+1); else break;
    n++;
  }
  return out;
}

function api_saveCalEntry(p){

function calendarReminders(){

    var e=find_('CalEntries','id',r.entryId); if(!e) return; var mins=num_(r.mins); if(!mins) return;
    // occurrences whose reminder time falls in the last ~16 minutes or is due now
    var occ=calOccurrences_(e,new Date(now.getTime()+(mins-16)*60000),new Date(now.getTime()+mins*60000));
    occ.forEach(function(o){ var key=r.entryId+'|'+r.userId+'|'+calFmt_(o); if(sent[key]||o<from) return;

      if(u.email) mail_(u.email,'Reminder: '+e.title,'Hello '+u.fullName+',\n\nThis is your reminder for:\n\n'+e.title+'\n'+when+(e.location?'\nLocation: '+e.location:'')+(e.link?'\nLink: '+e.link:'')+(e.note?'\n\n'+e.note:'')+'\n\n— '+(ev.name||APP_NAME));

  });
  return n;
}

function financeOut_(e){
  var ev=e.eventId, heads=jp_(e.budgetJSON), total=num_(e.budgetTotal), limit=num_(e.approvalLimit)||25000;
  var exps=readAll_('Expenses').filter(function(x){return x.eventId===ev;}).map(expOut_).sort(function(a,b){return String(b.date+b.addedAt).localeCompare(String(a.date+a.addedAt));});
  var inc=readAll_('Income').filter(function(x){return x.eventId===ev;}).map(function(x){ var o={}; SCHEMA.Income.forEach(function(k){o[k]=x[k];}); o.amount=num_(x.amount);
    var a=x.artistId?user_(x.artistId):null; o.artistName=a?a.fullName:''; return o; }).sort(function(a,b){return String(b.receivedOn).localeCompare(String(a.receivedOn));});
  var vendors=readAll_('Vendors').filter(function(v){return v.eventId===ev;}).map(function(v){
    var paid=0, approved=0; exps.forEach(function(x){ if(x.vendorId===v.vendorId){ if(x.status==='paid') paid+=x.amount; else if(x.status==='approved') approved+=x.amount; } });
    return {vendorId:v.vendorId,name:v.name,service:v.service,phone:v.phone,note:v.note,contractAmount:num_(v.contractAmount),paid:paid,approved:approved,due:num_(v.contractAmount)-paid};
  });
  var s={income:0,incomeByKind:{},paid:0,approvedUnpaid:0,requested:0,cash:0,bank:0,byHead:{},byTeam:{},byDay:{},advancesOut:0};
  inc.forEach(function(i){ s.income+=i.amount; s.incomeByKind[i.kind]=(s.incomeByKind[i.kind]||0)+i.amount; s[i.account==='cash'?'cash':'bank']+=i.amount; });
  exps.forEach(function(x){

      s.byHead[x.head]=(s.byHead[x.head]||0)+x.amount; var tn=x.teamName||'No team'; s.byTeam[tn]=(s.byTeam[tn]||0)+x.amount; s.byDay[x.date]=(s.byDay[x.date]||0)+x.amount;

    else if(x.status==='requested') s.requested+=x.amount;

  var names=heads.map(function(h){return h.name;});
  Object.keys(s.byHead).forEach(function(n){ if(names.indexOf(n)<0){ heads.push({name:n,alloc:0}); names.push(n);} });

  if(i.kind==='adopt' && !i.artistId) throw 'Choose the artist being adopted.';
  var row={kind:i.kind,sourceName:String(i.sourceName).trim(),sourcePhone:i.sourcePhone||'',sourceEmail:i.sourceEmail||'',amount:amt,account:i.account,mode:i.mode,
    receivedOn:i.receivedOn||now_().slice(0,10),artistId:i.kind==='adopt'?i.artistId:'',note:i.note||''};
  if(i.proof) row.proofId=saveFile_(SUB_BILLS,i.proof,'proof-'+Date.now()+'.jpg');
  if(i.incId){ updateWhere_('Income','incId',i.incId,row); log_(p.eventId,me,'Income edited',i.kind+' · '+row.sourceName+' ₹'+amt); }
  else{ row.incId=uid_('i'); row.eventId=p.eventId; row.createdBy=me.userId; row.createdAt=now_(); appendRow_('Income',row); log_(p.eventId,me,'Income added',i.kind+' · '+row.sourceName+' ₹'+amt); }
  return api_eventData({token:p.token,eventId:p.eventId});
}

/* ---- Super admin: people ---- */
function api_allUsers(p){
  var me=userByToken_(p.token); needSuper_(me);
  return readAll_('Users').map(function(u){ var o=pubUser_(u); o.mobile=u.mobile; o.email=u.email; o.isSuper=isSuper_(u); o.isDisabled=T_(u.isDisabled); o.lastLogin=u.lastLogin; o.createdAt=u.createdAt; return o; })
    .sort(function(a,b){return String(a.fullName).localeCompare(String(b.fullName));});
}
function api_setSuper(p){
  var me=userByToken_(p.token); needSuper_(me);
  var u=user_(p.userId); if(!u) throw 'Not found.';
  if(u.userId===me.userId && !p.value) throw 'You cannot remove your own super admin access.';
  updateWhere_('Users','userId',u.userId,{isSuper:p.value?'TRUE':'FALSE'});
  log_('',me,p.value?'Made super admin':'Removed super admin',u.fullName,u.userId);
  return api_allUsers({token:p.token});
}
function api_setDisabled(p){
  var me=userByToken_(p.token); needSuper_(me);
  var u=user_(p.userId); if(!u) throw 'Not found.'; if(u.userId===me.userId) throw 'You cannot disable yourself.';
  updateWhere_('Users','userId',u.userId,{isDisabled:p.value?'TRUE':'FALSE'});
  if(p.value) readAll_('Sessions').filter(function(s){return s.userId===u.userId;}).forEach(function(s){ deleteWhere_('Sessions','id',s.id); });
  log_('',me,p.value?'Account disabled':'Account enabled',u.fullName,u.userId);
  return api_allUsers({token:p.token});
}

function api_exportEventDoc(p){
  var me=userByToken_(p.token), e=find_('Events','eventId',p.eventId); if(!e) throw 'Not found.';
  need_(me,e.eventId,['admin']);
  var name=(e.name+' — Report').replace(/[\\\/:*?"<>|]/g,'-'), folder=subFolder_(SUB_REPORTS), it=folder.getFilesByName(name), doc;
  if(it.hasNext()){ doc=DocumentApp.openById(it.next().getId()); doc.getBody().clear(); }
  else{ doc=DocumentApp.create(name); DriveApp.getFileById(doc.getId()).moveTo(folder); }
  var b=doc.getBody(), roles=rolesOf_(e.eventId);
  b.appendParagraph(e.name).setHeading(DocumentApp.ParagraphHeading.TITLE);
  b.appendParagraph('Date: '+e.startDate+(e.endDate?' to '+e.endDate:'')+'   Venue: '+(e.venue||'-')+', '+(e.city||'')+'   Status: '+e.status);
  b.appendParagraph('Updated: '+Utilities.formatDate(new Date(),Session.getScriptTimeZone(),'dd MMM yyyy, hh:mm a'));
  [['Artists','artist'],['Volunteers','volunteer'],['Admins','admin']].forEach(function(g){
    var list=roles.filter(function(r){return r.role===g[1];}); if(!list.length) return;
    b.appendParagraph(g[0]+' ('+list.length+')').setHeading(DocumentApp.ParagraphHeading.HEADING2);
    var rows=[['Name','Mobile','State','Category','Status','Checked in']];
    list.forEach(function(r){ var u=user_(r.userId)||{}; rows.push([u.fullName||'',u.mobile||'',u.state||'',r.category||'',r.status||'',T_(r.checkedIn)?'Yes':'No']); });
    b.appendTable(rows);
  });
  var teams=teamsOut_(e.eventId,'admin',me);
  if(teams.length){ b.appendParagraph('Teams').setHeading(DocumentApp.ParagraphHeading.HEADING2);
    teams.forEach(function(t){ b.appendListItem(t.name+' — Head: '+(t.head?t.head.fullName+' (@'+t.head.username+')':'-')+(t.members.length?' — Members: '+t.members.map(function(m){return m.fullName;}).join(', '):'')); }); }
  var f=financeOut_(e);
  b.appendParagraph('Finance').setHeading(DocumentApp.ParagraphHeading.HEADING2);
  b.appendParagraph('Income: Rs '+f.summary.income+'   Paid expenses: Rs '+f.summary.paid+'   Net: Rs '+(f.summary.income-f.summary.paid)+'   Cash: Rs '+f.summary.cash+'   Bank: Rs '+f.summary.bank);

  var sch=scheduleOf_(e.eventId);
  if(sch.length){ b.appendParagraph('Schedule').setHeading(DocumentApp.ParagraphHeading.HEADING2); sch.forEach(function(s){ b.appendListItem((s.day||'')+' '+(s.time||'')+' — '+s.title+(s.detail?' ('+s.detail+')':'')); }); }
  doc.saveAndClose();
  return {url:doc.getUrl()};
}

/* ============================================================
 *  DAILY BACKUP (copy of the database sheet, last 14 kept)
 * ========================================================== */
function ensureBackupTrigger_(){
  try{
    var props=PropertiesService.getScriptProperties(); if(props.getProperty('BACKUP_TRIGGER')) return;
    if(!ScriptApp.getProjectTriggers().some(function(t){return t.getHandlerFunction()==='dailyBackup';}))
      ScriptApp.newTrigger('dailyBackup').timeBased().everyDays(1).atHour(2).create();
    props.setProperty('BACKUP_TRIGGER','1');
  }catch(e){}
}
function dailyBackup(){
  var ss=db_(), folder=subFolder_(SUB_BACKUPS);
  DriveApp.getFileById(ss.getId()).makeCopy('Backup '+Utilities.formatDate(new Date(),Session.getScriptTimeZone(),'yyyy-MM-dd'),folder);
  var files=[], it=folder.getFiles(); while(it.hasNext()) files.push(it.next());
  files.sort(function(a,b){return b.getDateCreated()-a.getDateCreated();});
  files.slice(BACKUP_KEEP).forEach(function(f){ f.setTrashed(true); });
  var old=new Date(Date.now()-864e5).toISOString();
  readAll_('Sessions').filter(function(s){return s.exp<old;}).forEach(function(s){ deleteWhere_('Sessions','id',s.id); });
}
