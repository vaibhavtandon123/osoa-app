function doPost(e){
  var out, lock=null;
  try{
    var req=JSON.parse((e&&e.postData&&e.postData.contents)||'{}');
    var fn=req.fn, arg=req.arg;
    if(!/^api_/.test(fn) || typeof globalThis[fn]!=='function') throw 'Unknown action';
    if(!READ_ONLY[fn]){ lock=LockService.getScriptLock(); lock.waitLock(25000); }
    out={ r: globalThis[fn](arg) };
  }catch(err){
    out={ e: (err&&err.message)?err.message:String(err) };
  }finally{
    if(lock){ try{ lock.releaseLock(); }catch(x){} }
  }
  return ContentService.createTextOutput(JSON.stringify(out)).setMimeType(ContentService.MimeType.JSON);
}

function ensureSheets_(ss){
  Object.keys(SCHEMA).forEach(function(name){
    var sh=ss.getSheetByName(name);
    if(!sh){ sh=ss.insertSheet(name); sh.getRange(1,1,1,SCHEMA[name].length).setValues([SCHEMA[name]]); }
    else{
      var have=sh.getRange(1,1,1,Math.max(1,sh.getLastColumn())).getValues()[0].map(String);
      SCHEMA[name].forEach(function(c){ if(have.indexOf(c)<0){ sh.getRange(1,sh.getLastColumn()+1).setValue(c); have.push(c); } });
    }
    sh.setFrozenRows(1);
    try{ sh.getRange(1,1,sh.getMaxRows(),sh.getMaxColumns()).setNumberFormat('@'); }catch(e){}
  });
  var def=ss.getSheetByName('Sheet1'); if(def && def.getLastRow()===0) ss.deleteSheet(def);
  if(readAll_('Programs').length===0) DEFAULT_PROGRAMS.forEach(function(p,i){ p.sort=i+1; p.logoId=''; appendRow_('Programs',p); });

}

function sheet_(name){ return db_().getSheetByName(name); }

function str_(v){ if(v instanceof Date) return Utilities.formatDate(v,Session.getScriptTimeZone(),"yyyy-MM-dd"); return String(v==null?'':v); }
function headerIdx_(sh){ var m={}; sh.getRange(1,1,1,sh.getLastColumn()).getValues()[0].forEach(function(c,i){ if(c!=='') m[String(c)]=i; }); return m; }
function readAll_(name){
  if(_cache[name]) return _cache[name];
  var sh=sheet_(name), last=sh.getLastRow(); if(last<2) return (_cache[name]=[]);
  var idx=headerIdx_(sh), vals=sh.getRange(2,1,last-1,sh.getLastColumn()).getValues();
  return (_cache[name]=vals.map(function(r){ var o={}; SCHEMA[name].forEach(function(c){ o[c]=(idx[c]!=null)?str_(r[idx[c]]):''; }); return o; }));
}
function appendRow_(name,obj){
  var sh=sheet_(name), idx=headerIdx_(sh), row=[]; for(var i=0;i<sh.getLastColumn();i++) row.push('');
  SCHEMA[name].forEach(function(c){ if(idx[c]!=null && obj[c]!==undefined && obj[c]!==null) row[idx[c]]=String(obj[c]); });
  sh.appendRow(row); delete _cache[name];
}
function updateWhere_(name,keyCol,keyVal,patch){
  var sh=sheet_(name), idx=headerIdx_(sh), last=sh.getLastRow(); if(last<2) return false;
  var rng=sh.getRange(2,1,last-1,sh.getLastColumn()), vals=rng.getValues(), ki=idx[keyCol];
  for(var r=0;r<vals.length;r++){
    if(String(vals[r][ki])===String(keyVal)){
      Object.keys(patch).forEach(function(c){ if(idx[c]!=null && patch[c]!==undefined) sh.getRange(r+2,idx[c]+1).setValue(String(patch[c]==null?'':patch[c])); });
      delete _cache[name]; return true;
    }
  }
  return false;
}
function deleteWhere_(name,keyCol,keyVal){
  var sh=sheet_(name), idx=headerIdx_(sh), last=sh.getLastRow(); if(last<2) return false;
  var ki=idx[keyCol], vals=sh.getRange(2,ki+1,last-1,1).getValues();
  for(var r=vals.length-1;r>=0;r--){ if(String(vals[r][0])===String(keyVal)){ sh.deleteRow(r+2); delete _cache[name]; return true; } }
  return false;
}

function hash_(t,s){ return Utilities.computeDigest(Utilities.DigestAlgorithm.SHA_256,t+'::'+s).map(function(b){return('0'+(b&0xFF).toString(16)).slice(-2);}).join(''); }
function digits_(s){ return String(s||'').replace(/[^0-9]/g,'').slice(-10); }
function isSuper_(u){ return !!u && (String(u.isSuper).toUpperCase()==='TRUE'); }

function api_signup(p){
  p=p||{}; var u=String(p.username||'').trim().toLowerCase();
  if(!p.fullName||!u||!p.password) throw 'Please fill name, username and password.';
  if(!/^[a-z0-9._]{3,30}$/.test(u)) throw 'Username: 3-30 letters, numbers, dot or underscore.';
  if(String(p.password).length<6) throw 'Password must be at least 6 characters.';
  if(digits_(p.mobile).length<10) throw 'Please enter a valid 10-digit mobile number.';
  var users=readAll_('Users');
  if(users.some(function(x){return String(x.username).toLowerCase()===u;})) throw 'That username is already taken.';
  if(users.some(function(x){return x.mobile && digits_(x.mobile)===digits_(p.mobile);})) throw 'That mobile number is already registered. Try logging in.';
  if(p.email && users.some(function(x){return String(x.email).toLowerCase()===String(p.email).trim().toLowerCase();})) throw 'That email is already registered.';
  var salt=Utilities.getUuid(), id=uid_('u');
  appendRow_('Users',{userId:id,fullName:String(p.fullName).trim(),username:u,passHash:hash_(p.password,salt),salt:salt,
    email:String(p.email||'').trim(),mobile:digits_(p.mobile),state:p.state||'',city:p.city||'',
    photoId:p.photo?savePhoto_(p.photo,id):'',createdAt:now_()});
  return api_login({username:u,password:p.password});
}
function api_login(p){
  p=p||{}; var k=String(p.username||'').trim().toLowerCase(), d=digits_(k);
  var user=readAll_('Users').filter(function(x){
    return String(x.username).toLowerCase()===k || (k.indexOf('@')>0 && String(x.email).toLowerCase()===k) || (d.length===10 && digits_(x.mobile)===d);
  })[0];
  if(!user) throw 'No account found with that username, email or mobile.';
  if(!user.passHash) throw 'Password not set yet. Use "Forgot password".';
  if(hash_(p.password,user.salt)!==user.passHash) throw 'Wrong password.';
  var token=Utilities.getUuid()+Utilities.getUuid(), exp=new Date(Date.now()+TOKEN_DAYS*864e5).toISOString();
  updateWhere_('Users','userId',user.userId,{tokenHash:hash_(token,user.salt),tokenExp:exp,lastLogin:now_()});
  return {token:token,me:meInfo_(find_('Users','userId',user.userId))};
}
function userByToken_(token){
  if(!token) throw 'Please log in again.';
  var users=readAll_('Users');
  for(var i=0;i<users.length;i++){ var x=users[i];
    if(x.tokenHash && hash_(token,x.salt)===x.tokenHash){
      if(x.tokenExp && new Date(x.tokenExp)<new Date()) throw 'Session expired. Please log in again.';
      return x;
    }
  }
  throw 'Please log in again.';
}

function api_updateProfile(p){
  var me=userByToken_(p.token), patch={};
  ['fullName','email','state','city','category','bio','instagram'].forEach(function(k){ if(p[k]!==undefined) patch[k]=String(p[k]||'').trim(); });
  if(p.mobile!==undefined){
    if(digits_(p.mobile).length<10) throw 'Please enter a valid 10-digit mobile number.';
    if(readAll_('Users').some(function(x){return x.userId!==me.userId && digits_(x.mobile)===digits_(p.mobile);})) throw 'That mobile number is used by another account.';
    patch.mobile=digits_(p.mobile);
  }
  if(patch.fullName==='') delete patch.fullName;
  if(p.photo) patch.photoId=savePhoto_(p.photo,me.userId);
  updateWhere_('Users','userId',me.userId,patch);
  return meInfo_(find_('Users','userId',me.userId));
}
function api_changePassword(p){
  var me=userByToken_(p.token);
  if(me.passHash && hash_(p.oldPassword||'',me.salt)!==me.passHash) throw 'Current password is wrong.';
  if(String(p.newPassword||'').length<6) throw 'New password must be at least 6 characters.';
  updateWhere_('Users','userId',me.userId,{passHash:hash_(p.newPassword,me.salt)});
  return true;
}
function api_forgotPassword(p){
  var email=String((p&&p.email)||'').trim().toLowerCase();
  var user=readAll_('Users').filter(function(x){return x.email && String(x.email).toLowerCase()===email;})[0];
  if(!user) throw 'No account with that email.';
  var code=''+Math.floor(100000+Math.random()*900000);
  CacheService.getScriptCache().put('reset-'+user.userId,code,900);
  MailApp.sendEmail(user.email,APP_NAME+' — password reset code','Hello '+user.fullName+',\n\nYour '+APP_NAME+' reset code is: '+code+'\nIt is valid for 15 minutes.\n\nYour username: '+user.username);
  return {userId:user.userId};
}
function api_resetPassword(p){
  var c=CacheService.getScriptCache(), code=c.get('reset-'+p.userId);
  if(!code||code!==String(p.code).trim()) throw 'Wrong or expired code.';
  if(String(p.password||'').length<6) throw 'Password must be at least 6 characters.';
  var user=find_('Users','userId',p.userId); if(!user) throw 'Account not found.';
  updateWhere_('Users','userId',user.userId,{passHash:hash_(p.password,user.salt)});
  c.remove('reset-'+p.userId);
  return api_login({username:user.username,password:p.password});
}

function api_image(id){
  if(!id) return '';
  try{
    var f=DriveApp.getFileById(id), app=appFolder_().getId(), ok=false, ps=f.getParents();
    while(ps.hasNext()){ var p=ps.next(); var pp=p.getParents(); if(pp.hasNext() && pp.next().getId()===app){ ok=true; break; } }
    if(!ok) return '';
    var b=f.getBlob(); return 'data:'+b.getContentType()+';base64,'+Utilities.base64Encode(b.getBytes());
  }catch(e){ return ''; }
}

/* Everything a logged-in person needs for one event, filtered by their role. */
function api_eventData(p){
  var me=userByToken_(p.token), e=find_('Events','eventId',p.eventId); if(!e) throw 'Event not found.';
  var role=roleOf_(me,e.eventId), out={event:eventOut_(e),role:role,program:find_('Programs','programId',e.programId),schedule:scheduleOf_(e.eventId),categories:ART_CATEGORIES};
  var roles=readAll_('Roles').filter(function(r){return r.eventId===e.eventId;});
  var myRole=roles.filter(function(r){return r.userId===me.userId;})[0];
  out.myRole=myRole?roleOut_(myRole,true):null;
  if(!role) return out;
  var anns=readAll_('Announcements').filter(function(a){return a.eventId===e.eventId && (role==='admin'||a.audience==='all'||a.audience===role);})
    .sort(function(a,b){return String(b.createdAt).localeCompare(String(a.createdAt));});
  out.announcements=anns;
  out.teams=teamsOut_(e.eventId,role,me);
  var adm={}; readAll_('Users').forEach(function(u){ if(isSuper_(u)) adm[u.userId]=u; });
  roles.forEach(function(r){ if(r.role==='admin'){ var u=find_('Users','userId',r.userId); if(u) adm[u.userId]=u; } });
  out.admins=Object.keys(adm).map(function(k){var u=adm[k];var o=pubUser_(u);o.mobile=u.mobile;return o;});
  out.myHeadTeams=headTeamIds_(me,e.eventId);
  out.teamSuggestions=TEAM_SUGGESTIONS;
  if(role==='admin'||role==='volunteer'){
    out.artists=roles.filter(function(r){return r.role==='artist' && (role==='admin'||r.status==='approved');}).map(function(r){return roleOut_(r,role==='admin');});
    var tasks=readAll_('Tasks').filter(function(t){return t.eventId===e.eventId;});
    var myT=myTeamIds_(me,e.eventId);
    out.tasks=(role==='admin'?tasks:tasks.filter(function(t){return t.assignedTo===me.userId||(t.teamId&&myT.indexOf(t.teamId)>=0)||(!t.assignedTo&&!t.teamId);})).map(taskOut_);
  }
  if(role==='admin'||isFinanceHead_(me,e.eventId)){ out.budget=budgetOut_(e); out.canSetBudget=(role==='admin'); }
  if(role==='admin'){
    out.team=roles.filter(function(r){return r.role!=='artist';}).map(function(r){return roleOut_(r,true);});
    out.activity=readAll_('Activity').filter(function(a){return a.eventId===e.eventId;}).slice(-60).reverse();
  }
  return out;
}

  return o;
}

/* ---- Artist quick register (uses the logged-in account) ---- */
function api_registerArtist(p){
  var me=userByToken_(p.token), e=find_('Events','eventId',p.eventId); if(!e) throw 'Event not found.';
  if(String(e.regOpen).toUpperCase()==='FALSE') throw 'Registrations are closed for this event.';
  var ex=readAll_('Roles').filter(function(r){return r.eventId===e.eventId&&r.userId===me.userId;})[0];
  if(ex && ex.role!=='artist') throw 'You are already part of this event as '+ex.role+'.';
  if(!p.category) throw 'Please choose your art category.';
  var arts=(ex?jp_(ex.artworkJSON):[]).filter(function(id){ return (p.keepArtworks||[]).indexOf(id)>=0; });
  (p.artworks||[]).slice(0,3).forEach(function(d,i){ arts.push(saveFile_(SUB_ART,d,'art-'+me.userId+'-'+Date.now()+'-'+i+'.jpg')); });
  arts=arts.slice(0,3);
  if(!arts.length) throw 'Please upload at least one artwork photo.';
  var prof={state:p.state,city:p.city,category:p.category,bio:p.bio,instagram:p.instagram};
  if(p.photo) prof.photo=p.photo;
  prof.token=p.token; api_updateProfile(prof);
  var row={category:p.category,bio:p.bio||'',artTitle:p.artTitle||'',artworkJSON:JSON.stringify(arts)};
  if(ex){ row.status=(ex.status==='rejected')?'pending':ex.status; updateWhere_('Roles','roleId',ex.roleId,row); log_(e.eventId,me,'Artist updated registration',''); }
  else{ row.roleId=uid_('r'); row.eventId=e.eventId; row.userId=me.userId; row.role='artist'; row.status='pending'; row.checkedIn='FALSE'; row.createdAt=now_();
    appendRow_('Roles',row); log_(e.eventId,me,'Artist registered',p.category); }
  return api_eventData({token:p.token,eventId:e.eventId});
}

/* ---- Admin: approve / reject artists (status + email; WhatsApp done from the app) ---- */

  if(u && u.email && p.status!=='pending'){
    try{
      var body='Hello '+u.fullName+',\n\n'+(p.status==='approved'
        ? 'Congratulations! Your artist registration for "'+e.name+'" has been approved.'
        : 'Thank you for registering for "'+e.name+'". Unfortunately your registration was not approved this time.')
        +(p.note?'\n\nNote: '+p.note:'')+'\n\nEvent: '+e.name+'\nDate: '+e.startDate+(e.venue?'\nVenue: '+e.venue:'')+'\n\n— '+(prog.deskName||APP_NAME);

function api_exportEventDoc(p){
  var me=userByToken_(p.token), e=find_('Events','eventId',p.eventId); if(!e) throw 'Not found.';
  need_(me,e.eventId,['admin']);
  var name=(e.name+' — Report').replace(/[\\\/:*?"<>|]/g,'-'), folder=subFolder_(SUB_REPORTS), it=folder.getFilesByName(name), doc;
  if(it.hasNext()){ doc=DocumentApp.openById(it.next().getId()); doc.getBody().clear(); }
  else{ doc=DocumentApp.create(name); DriveApp.getFileById(doc.getId()).moveTo(folder); }
  var b=doc.getBody(), roles=readAll_('Roles').filter(function(r){return r.eventId===e.eventId;});
  b.appendParagraph(e.name).setHeading(DocumentApp.ParagraphHeading.TITLE);
  b.appendParagraph('Date: '+e.startDate+(e.endDate?' to '+e.endDate:'')+'   Venue: '+(e.venue||'-')+', '+(e.city||'')+'   Status: '+e.status);
  b.appendParagraph('Updated: '+Utilities.formatDate(new Date(),Session.getScriptTimeZone(),'dd MMM yyyy, hh:mm a'));

    b.appendParagraph(g[0]+' ('+list.length+')').setHeading(DocumentApp.ParagraphHeading.HEADING2);

    list.forEach(function(r){ var u=find_('Users','userId',r.userId)||{}; rows.push([u.fullName||'',u.mobile||'',u.state||'',r.category||'',r.status||'',String(r.checkedIn).toUpperCase()==='TRUE'?'Yes':'No']); });
    b.appendTable(rows);
  });
  var teams=teamsOut_(e.eventId,'admin',me);
  if(teams.length){ b.appendParagraph('Teams').setHeading(DocumentApp.ParagraphHeading.HEADING2);
    teams.forEach(function(t){ b.appendListItem(t.name+' — Head: '+(t.head?t.head.fullName+' (@'+t.head.username+')':'-')+(t.members.length?' — Members: '+t.members.map(function(m){return m.fullName;}).join(', '):'')); }); }
  var bg=budgetOut_(e);
  if(bg.total||bg.expenses.length){ b.appendParagraph('Budget').setHeading(DocumentApp.ParagraphHeading.HEADING2);

  doc.saveAndClose();
  return {url:doc.getUrl()};
}

/* ============================================================
 *  TEAMS (departments) — admin picks teams + head; head manages members
 * ========================================================== */
function teamsOf_(eventId){ return readAll_('Teams').filter(function(t){return t.eventId===eventId;}).sort(function(a,b){return Number(a.sort||0)-Number(b.sort||0);}); }
function headTeamIds_(me,eventId){ return teamsOf_(eventId).filter(function(t){return t.headId===me.userId;}).map(function(t){return t.teamId;}); }
function myTeamIds_(me,eventId){
  var ids=headTeamIds_(me,eventId);
  readAll_('TeamMembers').forEach(function(m){ if(m.eventId===eventId && m.userId===me.userId && ids.indexOf(m.teamId)<0) ids.push(m.teamId); });
  return ids;
}
function isFinanceHead_(me,eventId){ return teamsOf_(eventId).some(function(t){return t.headId===me.userId && /financ/i.test(t.name);}); }
function teamsOut_(eventId,role,me){
  var mem=readAll_('TeamMembers').filter(function(m){return m.eventId===eventId;});
  return teamsOf_(eventId).map(function(t){
    var h=t.headId?find_('Users','userId',t.headId):null, canManage=(role==='admin'||t.headId===me.userId);
    var head=h?pubUser_(h):null; if(head) head.mobile=h.mobile;
    return {teamId:t.teamId,name:t.name,headId:t.headId,head:head,canManage:canManage,
      members:mem.filter(function(m){return m.teamId===t.teamId;}).map(function(m){ var u=find_('Users','userId',m.userId)||{}; var o=pubUser_(u); o.memberId=m.id; if(canManage) o.mobile=u.mobile; return o; })};
  });
}
function userByKey_(key){
  var k=String(key||'').trim().replace(/^@/,'').toLowerCase(), d=digits_(k);
  if(!k) throw 'Please enter a username.';
  var u=readAll_('Users').filter(function(x){ return String(x.username).toLowerCase()===k || (d.length===10 && digits_(x.mobile)===d); })[0];
  if(!u) throw 'No account with username "'+k+'". Ask them to create an account first.';
  return u;
}
function ensureVolunteer_(eventId,u,me){
  var ex=readAll_('Roles').filter(function(r){return r.eventId===eventId&&r.userId===u.userId;})[0];
  if(ex){ if(ex.role==='artist') throw u.fullName+' is registered as an artist in this event.'; return; }
  if(isSuper_(u)) return;
  appendRow_('Roles',{roleId:uid_('r'),eventId:eventId,userId:u.userId,role:'volunteer',status:'active',checkedIn:'FALSE',createdAt:now_(),artworkJSON:'[]'});
}

/* ============================================================
 *  BUDGET + EXPENSES — admin sets budget; admin or Finance head adds expenses
 * ========================================================== */
function budgetOut_(e){
  var heads=jp_(e.budgetJSON), total=Number(e.budgetTotal)||0;
  var exps=readAll_('Expenses').filter(function(x){return x.eventId===e.eventId;}).sort(function(a,b){return String(b.spentOn||b.addedAt).localeCompare(String(a.spentOn||a.addedAt));});
  var spentBy={}, spent=0; exps.forEach(function(x){ var a=Number(x.amount)||0; spent+=a; spentBy[x.head]=(spentBy[x.head]||0)+a; });
  var names=heads.map(function(h){return h.name;});
  Object.keys(spentBy).forEach(function(n){ if(names.indexOf(n)<0){ heads.push({name:n,alloc:0}); names.push(n);} });
  return {total:total,spent:spent,defaultHeads:BUDGET_HEADS,
    heads:heads.map(function(h){return {name:h.name,alloc:Number(h.alloc)||0,spent:spentBy[h.name]||0};}),
    expenses:exps.map(function(x){ var u=find_('Users','userId',x.addedBy); var o={}; SCHEMA.Expenses.forEach(function(k){o[k]=x[k];}); o.amount=Number(x.amount)||0; o.addedName=u?u.fullName:''; return o; })};
}
function api_setBudget(p){
  var me=userByToken_(p.token); need_(me,p.eventId,['admin']);
  var heads=(p.heads||[]).filter(function(h){return h&&String(h.name||'').trim();}).map(function(h){return {name:String(h.name).trim(),alloc:Math.max(0,Number(h.alloc)||0)};});
  var total=Math.max(0,Number(p.total)||0);
  updateWhere_('Events','eventId',p.eventId,{budgetTotal:total,budgetJSON:JSON.stringify(heads)});
  log_(p.eventId,me,'Budget set','Rs '+total);
  return api_eventData({token:p.token,eventId:p.eventId});
}
function canExpense_(me,eventId){ if(roleOf_(me,eventId)!=='admin' && !isFinanceHead_(me,eventId)) throw 'Only admins or the Finance head can do this.'; }
function api_saveExpense(p){
  var me=userByToken_(p.token), x=p.expense||{}; canExpense_(me,p.eventId);
  var amt=Number(x.amount); if(!(amt>0)) throw 'Please enter the amount.';
  if(!x.head) throw 'Please choose a budget head.';
  var row={head:x.head,amount:amt,note:x.note||'',paidBy:x.paidBy||'',spentOn:x.spentOn||''};
  if(x.bill) row.billId=saveFile_(SUB_BILLS,x.bill,'bill-'+Date.now()+'.jpg');
  if(x.expId){ var old=find_('Expenses','expId',x.expId); if(!old||old.eventId!==p.eventId) throw 'Not found.'; updateWhere_('Expenses','expId',x.expId,row); }
  else{ row.expId=uid_('x'); row.eventId=p.eventId; row.addedBy=me.userId; row.addedAt=now_(); appendRow_('Expenses',row); }
  log_(p.eventId,me,'Expense saved',x.head+' Rs '+amt);
  return api_eventData({token:p.token,eventId:p.eventId});
}
function api_deleteExpense(p){
  var me=userByToken_(p.token), x=find_('Expenses','expId',p.expId); if(!x) throw 'Not found.';
  canExpense_(me,x.eventId); deleteWhere_('Expenses','expId',x.expId); log_(x.eventId,me,'Expense deleted',x.head+' Rs '+x.amount);
  return api_eventData({token:p.token,eventId:x.eventId});
}
